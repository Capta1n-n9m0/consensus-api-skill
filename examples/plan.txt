# A batch plan: queries grouped by topic.
#
#   consensus.py batch -f plan.txt -o searches --dry-run    # check the tree first
#   consensus.py batch -f plan.txt -o searches --page-size 50
#   consensus.py batch -f plan.txt -o searches --resume     # after an interruption
#
# `@filters` before any group applies to every group; inside a group it applies to
# that group and overrides the global one. `::` gives a query a short folder name --
# worth using when the query is long, since the folder name is otherwise the
# truncated query and long paths are a real constraint on Windows.

@filters year_min=2015 human=true

progressive overload sarcopenia intervention :: progressive overload
protein supplementation with resistance training elderly :: protein co-supplementation
